\newglossaryentry{compressao-semantica}{
    name={Compressão semântica},
    description={estratégia de redução da representação textual de um artefato sem perda do conteúdo informacional essencial, recuperável por expansão determinística}
}

\newglossaryentry{transpilador}{
    name={Transpilador},
    description={programa que converte uma representação formal de entrada em outra representação formal de saída, preservando a semântica relevante do domínio}
}

\newglossaryentry{validacao-por-esquema}{
    name={Validação por esquema},
    description={verificação de conformidade de um documento estruturado em relação a uma especificação formal, como um XSD}
}
